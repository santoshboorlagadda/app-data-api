# Build/manifest files sent to the agent. Dockerfiles are INCLUDED — the upgrade must
# change the FROM base/build images too, and the agent can only edit what it can see.
MANIFEST_PATTERNS = [
    "*.csproj", "*.fsproj", "*.vbproj", "global.json", "Directory.Build.props", "nuget.config",
    "package.json", ".nvmrc", "pom.xml", "build.gradle",
    "Dockerfile", "Dockerfile.*", "*.Dockerfile", "docker-compose*.yml", ".dockerignore",
]


def _manifests(repo: Path) -> str:
    blocks, seen = [], set()
    for pat in MANIFEST_PATTERNS:
        for p in repo.rglob(pat):
            if ".git" in p.parts or "node_modules" in p.parts or p in seen or not p.is_file():
                continue
            seen.add(p)
            blocks.append(f"### {p.relative_to(repo)}\n{p.read_text(errors='ignore')[:6000]}")
    return "\n\n".join(blocks)

class UpgradeIn(BaseModel):
    app_id: str
    target_version: Optional[str] = None   # version number, e.g. "10" — required


@app.post("/api/upgrades")
async def start_upgrade(body: UpgradeIn, bg: BackgroundTasks) -> dict:
    """Upgrade ONE app. Base images come from base_images.json, same as Upgrade All."""
    a = APPS.get(body.app_id)
    if not a:
        raise HTTPException(404, "Unknown app")
    if not body.target_version:
        raise HTTPException(400, "target_version is required (e.g. \"10\").")

    imgs = catalog.image_set(a.type, body.target_version)
    if not imgs:
        raise HTTPException(400, f"No base image configured for {a.type} {body.target_version}.")
    if _same_version(a.current_version, body.target_version):
        raise HTTPException(400, f"{a.name} is already on {body.target_version}.")

    job = {"id": str(uuid.uuid4()), "kind": "upgrade", "app_id": a.id, "app_name": a.name,
           "target_version": f"{a.type} {body.target_version}", "state": "queued",
           "stages": [], "build_no": None, "images": imgs}
    JOBS[job["id"]] = job
    bg.add_task(_run_upgrade_task, job, {**a.model_dump(), "images": imgs,
                                         "target_version_num": body.target_version})
    return job
